Tworzenie aplikacji multimedialnych

Biblioteka SFML (ang. Simple and Fast Multimedia Library) to otwarte oprogramowanie (ang. open source), które zapewnia zestaw narzędzi do tworzenia aplikacji multimedialnych, szczególnie gier komputerowych i interaktywnych aplikacji graficznych. SFML oferuje funkcje do obsługi grafiki 2D, dźwięku, interakcji z użytkownikiem oraz programowania sieciowego.

Jest napisana w języku C++, co sprawia, że jest wydajna i łatwa w integracji z innymi bibliotekami oraz narzędziami programistycznymi. SFML jest dostępne na różne platformy, w tym Windows, Linux i macOS, co czyni ją popularnym wyborem dla twórców gier i aplikacji multimedialnych.

#include <SFML/Graphics.hpp>

auto main() -> int
{
    auto window = sf::RenderWindow(
        sf::VideoMode({800, 600}), "Test",
        sf::Style::Default, sf::State::Windowed,
        sf::ContextSettings{.antiAliasingLevel = 8}
    );

    while (window.isOpen())
    {
        while (auto const event = window.pollEvent())
        {
            if (event->is<sf::Event::Closed>())
            {
                window.close();
            }
        }

        window.display();
    }
}

Warto omówić niektóre z użytych wyżej elementów programu:

  • {800, 600}: Jest to rozmiar okna w pikselach. Pierwszy argument to szerokość, a drugi to wysokość. W tym przypadku okno ma rozmiar 800 pikseli na 600 pikseli.
  • "Test": Jest to tytuł okna, który zostanie wyświetlony na pasku tytułowym.
  • sf::Style::Default: Jest to styl okna, który definiuje wygląd i funkcje okna. sf::Style::Default oznacza standardowy styl okna, który obejmuje przycisk zamknięcia, zmniejszania i powiększania, a także pasek tytułowy. Gdyby użyć innego stylu, można by przykładowo stworzyć okno bez górnego paska zawierającego te przyciski (jak to bywa w większości gier).
  • sf::State::Windowed: Precyzuje, że aplikacja ma działać w oknie. Alternatywą byłby sf::State::Fullscreen. Warto zwrócić uwagę, że pełen ekran nie działałby dobrze ze sprecyzowaniem wielkości 800 na 600.
  • sf::ContextSettings{.antiAliasingLevel = 8}: Ustawia wybrane parametry związane z renderowaniem okna. Tutaj określamy wyłącznie poziom antyaliasingu (dodatkowe wygładzanie kształtów). Nie zaleca się ustawiania wartości powyżej 8 (skutkowałoby to minimalną poprawą przy wysokich kosztach obliczeniowych; wartość domyślna to 0, co oznacza brak antyaliasingu).

Późniejszy kod, czyli ta pętla w pętli, reprezentuje pętlę klatek, gdzie co klatkę przetwarzamy tyle zdarzeń (ang.event), ile pojawiło się od poprzedniej klatki. Bez tego okno nie będzie działało poprawnie. Nie jest to wyczerpujące wytłumaczenie, dlatego zaleca się przejść również oficjalny tutorial.

#include <SFML/Graphics.hpp>

auto main() -> int
{
    auto window = sf::RenderWindow(
        sf::VideoMode({800, 600}), "Test",
        sf::Style::Default, sf::State::Windowed,
        sf::ContextSettings{.antiAliasingLevel = 8}
    );

    auto hexagon = sf::CircleShape(50, 6);
    hexagon.setOutlineColor(sf::Color::Red);
    hexagon.setOutlineThickness(15);
    hexagon.setFillColor(sf::Color::Blue);
    hexagon.setPosition({100,100});

    while (window.isOpen())
    {
        while (auto const event = window.pollEvent())
        {
            if (event->is<sf::Event::Closed>())
            {
                window.close();
            }
        }

        //painting window with background color
        window.clear(sf::Color::White);
        //hexagon drawing
        window.draw(hexagon);
        //updating window display
        window.display();
    }
}

#include <SFML/Graphics.hpp>

auto main() -> int
{
    auto window = sf::RenderWindow(
        sf::VideoMode({800, 600}), "Test",
        sf::Style::Default, sf::State::Windowed,
        sf::ContextSettings{.antiAliasingLevel = 8}
    );

    auto hexagon = sf::CircleShape(50, 6);

    while (window.isOpen())
    {
        while (auto const event = window.pollEvent())
        {
            if (event->is<sf::Event::Closed>())
            {
                window.close();
            }
        }

        window.clear(sf::Color::Red);

        for (auto y = 0; y < 3; y++)
        {
            for (auto x = 0; x < 2; x++)
            {
                if (y % 2 == 0) hexagon.setFillColor(sf::Color::Black);
                else hexagon.setFillColor(sf::Color::White);

                hexagon.setPosition({
                    (y % 2 ? 280 : 230) + x * 100.f, 100 + y * 85.f
                });

                window.draw(hexagon);
            }
        }

        window.display();
    }
}

#include <SFML/Graphics.hpp>

auto main() -> int
{
    auto window = sf::RenderWindow(
        sf::VideoMode({800, 600}), "Test",
        sf::Style::Default, sf::State::Windowed,
        sf::ContextSettings{.antiAliasingLevel = 8}
    );

    auto const arial = sf::Font("../fonts/arial.ttf");
    auto text        = sf::Text(arial, "Click to change", 24);
    text.setPosition({300, 250});

    auto visibleButton = sf::RectangleShape({300, 50});
    visibleButton.setPosition({250, 300});
    visibleButton.setFillColor(sf::Color::Green);

    auto buttonText = sf::Text(arial, "click", 24);
    buttonText.setPosition({350, 310});

    auto const onClose = [&window](sf::Event::Closed const&) {
        window.close();
    };

    auto const onMouseClicked =
            [&visibleButton, &buttonText]
            (sf::Event::MouseButtonPressed const& e)
    {
        if (e.button == sf::Mouse::Button::Left)
        {
            auto const clickedOnButton = visibleButton
                    .getGlobalBounds()
                    .contains(static_cast<sf::Vector2f>(e.position));
            if (clickedOnButton)
            {
                if (buttonText.getString() == "click")
                {
                    buttonText.setString("clicked!");
                }
                else
                {
                    buttonText.setString("click");
                }
            }
        }
    };

    while (window.isOpen())
    {
        window.handleEvents(onClose, onMouseClicked);

        window.clear(sf::Color::Black);
        window.draw(text);
        window.draw(visibleButton);
        window.draw(buttonText);
        window.display();
    }
}

#include <SFML/Graphics.hpp>

auto main() -> int
{
    using namespace sf;
    
    auto window = RenderWindow(
        VideoMode({800, 600}), "Test",
        Style::Default, State::Windowed,
        ContextSettings{.antiAliasingLevel = 8}
    );

    // Creating a shape (in this case a square)
    auto shape = RectangleShape(Vector2f(100.f, 100.f));
    shape.setFillColor(Color::Blue);

    // Setting the initial speed of the shape
    auto velocity = Vector2f(0.02f, 0.02f);

    while (window.isOpen())
    {
        while (auto const event = window.pollEvent())
        {
            if (event->is<Event::Closed>()) window.close();
        }

        auto const position = shape.getPosition();
        // Bounce the shape off the edge of the window
        if (position.x < 0 || position.x + shape.getSize().x > window.getSize().x)
            velocity.x = -velocity.x;
        if (position.y < 0 || position.y + shape.getSize().y > window.getSize().y)
            velocity.y = -velocity.y;

        shape.move(velocity);

        // Drawing the shape on the screen
        window.clear();
        window.draw(shape);
        window.display();
    }
}

Pierwsza gra - PONG

Oto przedstawienie implementacji prostej wersji gry PONG. W tej grze gracz konfrontuje się z komputerem. Sterowanie paletką gracza odbywa się za pomocą klawiszy S i W, pozwalając mu na reakcję na ruchy piłeczki, podczas gdy komputer automatycznie dostosowuje położenie swojej paletki w celu skutecznego odbicia piłeczki. Kod ten ilustruje nie tylko obsługę klawiszy i ustawianie limitu klatek na sekundę, ale także wykorzystanie funkcji .move() oraz .intersects() do płynnego poruszania się obiektów i wykrywania kolizji.

#include <SFML/Graphics.hpp>

#include <fmt/format.h>

auto main() -> int
{
    using namespace sf;

    auto window = RenderWindow(
        VideoMode({800, 600}), "Test",
        Style::Default, State::Windowed,
        ContextSettings{.antiAliasingLevel = 8}
    );

    window.setFramerateLimit(240);
    //ball position and parameters
    auto ball = CircleShape(10);
    ball.setFillColor(Color::Green);
    ball.setPosition({400, 300});
    auto ballVelocity = Vector2f(1, 1);

    //parameters of first player
    auto player = RectangleShape(Vector2f(20, 100));
    player.setFillColor(Color::Red);
    player.setPosition({40, 250});

    //parameters for computer
    auto computer = RectangleShape(Vector2f(20, 100));
    computer.setFillColor(Color::Blue);
    computer.setPosition({740, 250});

    // players scores
    auto playerScore = 0;
    auto computerScore = 0;
    auto font = Font("../fonts/arial.ttf");
    auto scoreText = Text(font, "", 25);
    scoreText.setFillColor(Color::White);
    scoreText.setPosition({250, 15});

    auto playerMovingUp = false;
    auto playerMovingDown = false;

    while (window.isOpen()) {
        while (auto const event = window.pollEvent()) {
            if (event->is<Event::Closed>()) window.close();

            if (auto const e = event->getIf<Event::KeyPressed>())
            {
                if (e->code == Keyboard::Key::W) playerMovingUp = true;
            }

            if (auto const e = event->getIf<Event::KeyPressed>())
            {
                if (e->code == Keyboard::Key::S) playerMovingDown = true;
            }

            if (auto const e = event->getIf<Event::KeyReleased>())
            {
                if (e->code == Keyboard::Key::W) playerMovingUp = false;
            }

            if (auto const e = event->getIf<Event::KeyReleased>())
            {
                if (e->code == Keyboard::Key::S) playerMovingDown = false;
            }
        }

        // player movements (W,S)
        if (playerMovingUp && player.getPosition().y > 0)
        {
            player.move({0, -3});
        }
        else if (playerMovingDown && player.getPosition().y < window.getSize().y - player.getSize().y)
        {
            player.move({0, 3});
        }

        // computer movements
        if (ball.getPosition().y < computer.getPosition().y)
        {
            computer.move({0, -2});
        }
        else if (ball.getPosition().y > computer.getPosition().y + 100)
        {
            computer.move({0, 2});
        }

        //ball move
        ball.move(ballVelocity);

        // Bounce the ball off the top and bottom of the window
        if (ball.getPosition().y <= 0 || ball.getPosition().y >= window.getSize().y - ball.getRadius())
        {
            ballVelocity.y *= -1;
        }
        // Bouncing the ball off a player's pallet
        if (player.getGlobalBounds().findIntersection(ball.getGlobalBounds()))
        {
            ballVelocity.x *= -1;
        }

        // Bounce the ball off a computer's pallet
        if (computer.getGlobalBounds().findIntersection(ball.getGlobalBounds()))
        {
            ballVelocity.x *= -1;
        }

        // Checking whether the ball has crossed the left or right edge of the window
        auto const computerScoredPoint = ball.getPosition().x <= 0;
        auto const playerScoredPoint   = ball.getPosition().x >= window.getSize().x;

        if (computerScoredPoint) computerScore += 1;
        if (playerScoredPoint) playerScore += 1;

        if (computerScoredPoint or playerScoredPoint)
        {
            auto const screenMiddle = Vector2f(
                window.getSize().x / 2.0f, window.getSize().y / 2.0f
            );
            ball.setPosition(screenMiddle);
            ballVelocity.x *= -1;
        }

        //real-time display of the score
        scoreText.setString(fmt::format("Player: {}   Computer: {}", playerScore, computerScore));
        // drawing the elements
        window.clear();
        window.draw(player);
        window.draw(computer);
        window.draw(ball);
        window.draw(scoreText);
        window.display();
    }
}
